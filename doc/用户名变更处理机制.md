# 用户名变更处理机制

本文档详细说明 TMD (Twitter Media Downloader) 如何处理 Twitter/X 用户修改用户名后自动重命名文件夹的功能。

## 功能概述

当 Twitter/X 用户修改其用户名（screen_name）或显示名称（name）后，TMD 在下次下载该用户内容时会自动检测变更并重命名对应的文件夹，同时保持所有数据的完整性。

## 核心实现

### 1. 用户名变更检测

**位置**: [`internal/database/user_sync.go`](../internal/database/user_sync.go)

```go
func SyncUser(db *sqlx.DB, userId uint64, name string, screenName string, isProtected bool, friendsCount int, accessible bool) error {
	renamed := false
	isNew := false

	usrdb, err := GetUserById(db, userId)
	if err != nil {
		return err
	}
	if usrdb == nil {
		isNew = true
	} else {
		renamed = usrdb.Name != name || usrdb.ScreenName != screenName
	}

	// 改名：在 UPSERT 之前记录旧名称，确保 RecordUserPreviousName 失败时
	// 用户表中仍保留旧名称，不丢失改名历史。
	if renamed {
		if err := RecordUserPreviousName(db, userId, usrdb.Name, usrdb.ScreenName); err != nil {
			return err
		}
	}

	// 使用 UPSERT（INSERT ... ON CONFLICT）消除并发下的 TOCTOU 竞态：
	// 两个 goroutine 同时判断用户不存在时，后一个 INSERT 不再因 PK 冲突而失败。
	stmt := `INSERT INTO users(id, screen_name, name, protected, friends_count, is_accessible)
		VALUES (?, ?, ?, ?, ?, ?)
		ON CONFLICT(id) DO UPDATE SET
			screen_name=excluded.screen_name,
			name=excluded.name,
			protected=excluded.protected,
			friends_count=excluded.friends_count,
			is_accessible=excluded.is_accessible`
	if _, err := db.Exec(stmt, userId, screenName, name, isProtected, friendsCount, accessible); err != nil {
		return fmt.Errorf("failed to upsert user %d (%s): %w", userId, screenName, err)
	}

	// 新用户：首次同步记录当前名称作为基线。这样后续改名时
	// user_previous_names 表中才包含完整改名链（含原始名称）。
	if isNew {
		if err := RecordUserPreviousName(db, userId, name, screenName); err != nil {
			return fmt.Errorf("failed to record initial name for new user %d (%s): %w", userId, screenName, err)
		}
	}
	return nil
}
```

### 2. 文件夹重命名逻辑

**位置**: [`internal/entity/sync.go`](../internal/entity/sync.go)

```go
func Sync(e Entity, expectedName string) error {
    if !e.Recorded() {
        return e.Create(expectedName)  // 新用户：创建文件夹
    }

    name, err := e.Name()
    if err != nil {
        return err
    }
    if name != expectedName {
        return e.Rename(expectedName)  // 用户名变更：重命名文件夹
    }

    p, err := e.Path()
    if err != nil {
        return err
    }
    return os.MkdirAll(p, 0755)
}
```

### 3. 重命名具体实现

**位置**: [`internal/entity/user.go`](../internal/entity/user.go)

```go
func (ue *UserEntity) Rename(title string) error {
    if !ue.created {
        return fmt.Errorf("user entity [%s:%d] was not created", ue.record.ParentDir, ue.record.UserId)
    }

    old, _ := ue.Path()
    newPath := filepath.Join(filepath.Dir(old), title)

    // 使用系统调用重命名文件夹
    err := os.Rename(old, newPath)
    if os.IsNotExist(err) {
        // 边界情况：原文件夹被手动删除，创建新文件夹
        err = os.Mkdir(newPath, 0755)
    }
    if err != nil && !os.IsExist(err) {
        return err
    }

    // 更新数据库记录
    ue.record.Name = title
    return database.UpdateUserEntity(ue.db, ue.record)
}
```

### 4. Profile 下载时的同步处理

**位置**: `internal/downloading/profile/downloader.go`

Profile 下载使用相同的重命名逻辑：

```go
func (pd *ProfileDownloader) syncUserDirectory(profile *ProfileInfo, userTitle, screenName string) (string, error) {
	if err := database.SyncUser(pd.db, profile.ID, profile.Name, screenName, profile.Protected, 0, true); err != nil {
		log.Errorf("[profile] User sync failed screen_name=%s uid=%d error=%q", screenName, profile.ID, err.Error())
		return "", err
	}

	entity, err := database.LocateUserEntity(pd.db, profile.ID, pd.storage.usersBasePath)
	if err != nil {
		log.Errorf("[profile] User entity locate failed screen_name=%s uid=%d parent_dir=%q error=%q", screenName, profile.ID, logging.Path(pd.storage.usersBasePath), err.Error())
		return "", err
	}
	expectedTitle := userTitle

	if entity == nil {
		// 新用户：创建目录 + 记录实体
		entity = &database.UserEntity{UserId: profile.ID, ParentDir: pd.storage.usersBasePath, Name: expectedTitle}
		userDir := filepath.Join(pd.storage.usersBasePath, expectedTitle)
		if err := os.MkdirAll(userDir, 0755); err != nil {
			return "", err
		}
		if err := database.CreateUserEntity(pd.db, entity); err != nil {
			return "", err
		}
		log.Infof("[profile] User directory created screen_name=%s path=%q", screenName, logging.Path(userDir))
		return ensureProfileDirs(userDir)
	}

	oldUserDir, err := entity.Path()
	if err != nil {
		return "", err
	}
	if entity.Name == expectedTitle {
		// 用户名未变更，确保目录存在
		return ensureProfileDirs(oldUserDir)
	}

	// 用户名变更，重命名目录（原目录被手动删除时自动新建）
	newUserDir := filepath.Join(pd.storage.usersBasePath, expectedTitle)
	if err := os.Rename(oldUserDir, newUserDir); err != nil {
		if os.IsNotExist(err) {
			if mkdirErr := os.MkdirAll(newUserDir, 0755); mkdirErr != nil {
				return "", mkdirErr
			}
		} else {
			return "", err
		}
	}

	// 更新数据库
	entity.Name = expectedTitle
	if err := database.UpdateUserEntity(pd.db, entity); err != nil {
		return "", err
	}
	log.Infof("[profile] User directory renamed screen_name=%s old_path=%q new_path=%q", screenName, logging.Path(oldUserDir), logging.Path(newUserDir))
	return ensureProfileDirs(newUserDir)
}
```

## 符号链接同步

对于列表中的用户（通过符号链接引用），系统会自动更新所有指向该用户的符号链接。

**位置**: [`internal/downloading/entity.go`](../internal/downloading/entity.go)

```go
func updateUserLink(lnk *database.UserLink, db *sqlx.DB, path string) error {
    name := filepath.Base(path)

    linkpath, err := lnk.Path(db)
    if err != nil {
        return err
    }
    path, err = filepath.Abs(path)
    if err != nil {
        return err
    }

    linkDir := filepath.Dir(linkpath)
    if err := os.MkdirAll(linkDir, 0755); err != nil {
        return err
    }

    if lnk.Name == name {
        // 用户未改名，确保链接存在（陈旧链接会被替换）
        return ensureUserSymlink(path, linkpath)
    }

    // 用户改名，更新链接
    newlinkpath := filepath.Join(linkDir, name)

    // 删除旧链接
    if err = os.RemoveAll(linkpath); err != nil {
        return err
    }
    // 创建新链接
    if err = ensureUserSymlink(path, newlinkpath); err != nil {
        return err
    }
    // 更新数据库
    if err = database.UpdateUserLink(db, lnk.Id, name); err != nil {
        return err
    }

    lnk.Name = name
    return nil
}
```

`ensureUserSymlink` 在目标已存在但指向旧路径时调用 `replaceStaleUserSymlink` 替换陈旧链接。

## 数据库表结构

### users 表
存储用户基本信息，用于检测用户名变更。

```sql
CREATE TABLE users (
    id INTEGER NOT NULL, 
    screen_name VARCHAR NOT NULL, 
    name VARCHAR NOT NULL, 
    protected BOOLEAN NOT NULL, 
    friends_count INTEGER NOT NULL, 
    is_accessible BOOLEAN NOT NULL DEFAULT 1,
    PRIMARY KEY (id), 
    UNIQUE (screen_name)
);
```

### user_entities 表
存储用户文件夹路径信息。

```sql
CREATE TABLE user_entities (
    id INTEGER NOT NULL, 
    user_id INTEGER NOT NULL, 
    name VARCHAR NOT NULL,           -- 文件夹名称
    latest_release_time DATETIME,    -- 最新推文时间
    parent_dir VARCHAR COLLATE NOCASE NOT NULL, 
    media_count INTEGER,
    PRIMARY KEY (id), 
    UNIQUE (user_id, parent_dir), 
    FOREIGN KEY(user_id) REFERENCES users (id)
);
```

### user_previous_names 表
记录用户的历史用户名。

```sql
CREATE TABLE user_previous_names (
    id INTEGER NOT NULL, 
    user_id INTEGER NOT NULL, 
    screen_name VARCHAR NOT NULL, 
    name VARCHAR NOT NULL, 
    record_date DATE NOT NULL, 
    PRIMARY KEY (id), 
    FOREIGN KEY(user_id) REFERENCES users (id)
);
```

### user_links 表
存储列表中用户的符号链接信息。

```sql
CREATE TABLE user_links (
    id INTEGER NOT NULL,
    user_id INTEGER NOT NULL, 
    name VARCHAR NOT NULL, 
    parent_lst_entity_id INTEGER NOT NULL,
    PRIMARY KEY (id),
    UNIQUE (user_id, parent_lst_entity_id),
    FOREIGN KEY(user_id) REFERENCES users (id), 
    FOREIGN KEY(parent_lst_entity_id) REFERENCES lst_entities (id)
);
```

## 边界情况处理

### 1. 原文件夹被手动删除

```go
err := os.Rename(old, newPath)
if os.IsNotExist(err) {
    err = os.Mkdir(newPath, 0755)  // 自动创建新文件夹
}
```

**处理方式**: 自动创建新文件夹，继续正常工作。

### 2. 新文件夹名已存在

```go
if err != nil && !os.IsExist(err) {
    return err
}
```

**处理方式**: 忽略 `os.IsExist` 错误继续执行。这种情况在实际中极少发生，因为 Twitter screen_name 是唯一的。

### 3. 符号链接权限问题

在 Windows 上创建符号链接需要管理员权限。如果权限不足，程序会记录警告但不会中断下载：

```go
if symlinkWarnCount > 0 {
    log.Warnf("[batch] Symlink permission denied suppressed=%d hint=run_as_admin", symlinkWarnCount)
}
```

（首次发生时会先输出一条 Debug 日志 `[batch] Symlink create/update failed user=... reason=permission_denied suppressing=true`，之后计数抑制，见 `internal/downloading/batch_download.go`。）

## 总结

| 功能 | 状态 | 说明 |
|------|------|------|
| 检测用户名变更 | ✅ | 通过比对数据库记录和 API 返回 |
| 自动重命名文件夹 | ✅ | 使用 `os.Rename` 系统调用 |
| 更新数据库记录 | ✅ | 更新 `user_entities` 表 |
| 记录历史用户名 | ✅ | 写入 `user_previous_names` 表 |
| 同步符号链接 | ✅ | 更新列表中的用户链接 |
| Profile 下载支持 | ✅ | 同样支持重命名 |

**结论**: 用户修改 Twitter/X 用户名后，下次下载时程序会自动检测变更并重命名对应的文件夹，同时保持所有数据完整性。
